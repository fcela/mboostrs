PROCESSES = ('absolute', 'monotone', 'bell', 'constant', 'heavy_tail')


def draw(process, seed, n):
    rng = np.random.default_rng(seed)
    x1, x2 = rng.uniform(-1, 1, (2, n))
    scale = {'absolute': lambda: .3 + 1.5 * np.abs(x1),
             'monotone': lambda: np.exp(.7 * x1),
             'bell': lambda: .3 + 1.5 * np.exp(-8 * x1*x1),
             'constant': lambda: np.ones(n), 'heavy_tail': lambda: np.ones(n)}[process]()
    noise = rng.standard_t(3, n) / np.sqrt(3) if process == 'heavy_tail' else rng.normal(size=n)
    mean = np.sin(2*np.pi*x1) + .5*x2
    return {'x1': x1, 'x2': x2, 'y': mean + scale*noise}, scale


def score(lower, upper, test, scale, process):
    y = test['y']
    lo, hi = np.asarray(lower), np.asarray(upper)
    assert lo.shape == hi.shape == y.shape and np.all(np.isfinite(lo + hi))
    assert np.all(hi >= lo)
    width = hi-lo
    covered = (lo <= y) & (y <= hi)
    low, high = np.quantile(scale, [1/3, 2/3])
    z = stats.t.ppf(.95, 3)/np.sqrt(3) if process == 'heavy_tail' else stats.norm.ppf(.95)
    corr = None if np.ptp(width) < 1e-8*width.mean() or np.ptp(scale) == 0 else float(
        stats.spearmanr(width, scale).statistic)
    return dict(coverage=float(covered.mean()), width=float(width.mean()),
        coverage_low=float(covered[scale <= low].mean()), coverage_high=float(covered[scale >= high].mean()),
        width_scale_rank=corr, oracle_width=float(2*z*scale.mean()),
        interval_score=float(np.mean(width + 20*np.maximum(lo-y, 0) + 20*np.maximum(y-hi, 0))),
        prediction_digest=data_digest([lo, hi]))


def fit_band(method, process, seed, train, test):
    new = {key: test[key] for key in ('x1', 'x2')}
    if method == 'discover':
        result = mb.discover(train, 'y', newdata=new, seed=seed, level=.9)
        return result.predictions['lower'], result.predictions['upper'], dict(
            mean_formula=result.formula, spread_formula=result.spread_formula,
            spread_log=[row for row in result.log if row['stage'] == 'spread'],
            discovery_log=result.log, provenance=result.provenance)
    if method == 'known':
        formula = 'y ~ bbs(x1, df=4) + bols(x2)'
        varying = process in ('absolute', 'monotone', 'bell')
        if varying:
            model = mb.gamboostLSS({'mu': formula, 'sigma': 'y ~ bbs(x1, df=4)'}, train,
                                  family=mb.GaussianLSS(), control=mb.boost_control(100))
        else:
            model = mb.gamboost(formula, train, family=mb.Gaussian(), control=mb.boost_control(100))
        ci = mb.conformal_interval(model, newdata=new, level=.9, seed=seed,
                                  method='distributional' if varying else 'split')
        return ci.lower, ci.upper, {'n_calibration': ci.n_calibration}
    from catboost import CatBoostRegressor
    from lightgbm import LGBMRegressor
    from sklearn.ensemble import GradientBoostingRegressor
    x = np.column_stack([train['x1'], train['x2']])
    xt = np.column_stack([test['x1'], test['x2']])
    y = train['y']
    order = np.random.default_rng(seed + 10000).permutation(len(y))
    fit, calibration = order[:500], order[500:]
    brand, mode = method.split('_')
    def estimator(alpha=None):
        if brand == 'catboost':
            return CatBoostRegressor(iterations=500, depth=6, learning_rate=.05, verbose=False,
                thread_count=1, allow_writing_files=False, random_seed=seed,
                loss_function='RMSE' if alpha is None else f'Quantile:alpha={alpha}')
        if brand == 'lightgbm':
            return LGBMRegressor(n_estimators=300, num_leaves=15, learning_rate=.05, verbosity=-1,
                n_jobs=1, random_state=seed, objective='regression' if alpha is None else 'quantile',
                **({} if alpha is None else {'alpha': alpha}))
        return GradientBoostingRegressor(n_estimators=300, max_depth=3, learning_rate=.05,
            random_state=seed, loss='quantile', alpha=alpha)
    if mode == 'split':
        model = estimator().fit(x[fit], y[fit])
        mu_cal, mu_test = model.predict(x[calibration]), model.predict(xt)
        residual = np.abs(y[calibration] - mu_cal)
        lo, hi = mu_test, mu_test
    else:
        low, high = estimator(.05).fit(x[fit], y[fit]), estimator(.95).fit(x[fit], y[fit])
        residual = np.maximum(low.predict(x[calibration])-y[calibration],
                              y[calibration]-high.predict(x[calibration]))
        lo, hi = low.predict(xt), high.predict(xt)
    q = np.sort(residual)[min(int(np.ceil((len(calibration)+1)*.9)), len(calibration))-1]
    return lo-q, hi+q, {'n_fit': 500, 'n_calibration': 500, 'settings': 'PROTOCOL.md fixed controls'}


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument('--phase', choices=('baseline', 'fixed', 'diagnostic'), required=True)
    args = parser.parse_args()
    seeds = range(6) if args.phase == 'diagnostic' else range(562100, 562120)
    processes = ('absolute',) if args.phase == 'diagnostic' else PROCESSES
    methods = ('discover',) if args.phase in ('fixed', 'diagnostic') else (
        'discover', 'known', 'catboost_split', 'catboost_cqr', 'lightgbm_split', 'lightgbm_cqr', 'sklearn_cqr')
    output = DIRECTORY / f'{args.phase}.json'
    if output.exists():
        raise FileExistsError(f'Refusing to replace completed evidence: {output}')
    document = evidence_document({'id': 'nonmonotone-spread/1', 'phase': args.phase,
        'sha256': hashlib.sha256((DIRECTORY/'PROTOCOL.md').read_bytes()).hexdigest()},
        {'seeds': list(seeds), 'independent_test_seed_offset': 50000}, [Path(__file__)])
    source = Path(mb.__file__).parent
    document['loaded_sources'] = {str(source/name): hashlib.sha256((source/name).read_bytes()).hexdigest()
                                  for name in ('discover.py', 'explore.py')}
    for process in processes:
        for seed in seeds:
            train, _ = draw(process, seed, 1000)
            test, scale = draw(process, seed+50000, 2000)
            for method in methods:
                start = time.perf_counter()
                row = dict(process=process, seed=seed, method=method, phase=args.phase,
                    input_digest=data_digest([train['x1'], train['x2'], train['y'],
                                              test['x1'], test['x2'], test['y']]))
                with warnings.catch_warnings(record=True) as caught:
                    warnings.simplefilter('always')
                    try:
                        lo, hi, details = fit_band(method, process, seed, train, test)
                        row.update(status='ok', details=details, **score(lo, hi, test, scale, process))
                    except Exception:
                        row.update(status='failed', traceback=traceback.format_exc())
                row.update(warnings=[str(w.message) for w in caught], elapsed=time.perf_counter()-start)
                document['results'].append(row)
                write_evidence(document, output)
            print(json.dumps({'process': process, 'seed': seed, 'records': len(document['results'])}), flush=True)


if __name__ == '__main__':
    main()
